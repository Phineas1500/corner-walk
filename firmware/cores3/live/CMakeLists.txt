cmake_minimum_required(VERSION 3.20)
set(stage "$ENV{WALK_COMPONENTS}")
set(peer "$ENV{LIVE_PEER_DEPS_DIR}")
set(sdk "$ENV{WALK_LIVE_SDK}")
set(EXTRA_COMPONENT_DIRS "${stage}" "${peer}/mbedtls" "${sdk}/components/live" "${sdk}/components/live_peer")
set(COMPONENTS main console)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(corner_walk_live)
idf_build_set_property(COMPILE_OPTIONS "-g0" APPEND)
idf_build_set_property(COMPILE_DEFINITIONS "ARDUINO_USB_MODE=1;ARDUINO_USB_CDC_ON_BOOT=1;BOARD_HAS_PSRAM;WALK_LIVE" APPEND)
idf_component_get_property(arduino_dir arduino COMPONENT_DIR)
set_property(SOURCE "${arduino_dir}/cores/esp32/esp32-hal-i2c-slave.c" DIRECTORY "${arduino_dir}"
    APPEND PROPERTY COMPILE_OPTIONS -include "${CMAKE_CURRENT_LIST_DIR}/i2c_idf53.h")
idf_component_get_property(arduino_lib arduino COMPONENT_LIB)
target_link_libraries(${arduino_lib} PUBLIC idf::esp_netif idf::esp_wifi idf::esp_timer)
